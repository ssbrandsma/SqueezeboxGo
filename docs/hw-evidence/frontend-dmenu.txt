#!/bin/bash

#export SDL_NOMOUSE=1
export LD_LIBRARY_PATH=/usr/lib32:/usr/lib:/mnt/vendor/lib

ACT="/tmp/.next"
CMD="/mnt/vendor/bin/dmenu.bin"
MISCBIN="/mnt/mmc/dmenu.bin"

if [ -f "/mnt/vendor/muos1.ini" ];then
    MISCBIN="/mnt/vendor/bin/muos1.bin"
elif [ -f "/mnt/vendor/muos2.ini" ];then
    MISCBIN="/mnt/vendor/bin/muos2.bin"
fi

#add for video
if [ -f "/tmp/vpRun.ini" ];then
	MISCBIN="/mnt/vendor/bin/video/vp.dge"
fi

if [ -f  $MISCBIN ]
then
	CMD=$MISCBIN
fi

#touch "$ACT"

function app_scheduling()
{
	[ -f /mnt/mod/ctrl/configs/led_exit.cfg ] && cat /mnt/mod/ctrl/configs/led_exit.cfg > /sys/class/power_supply/axp2202-battery/work_led
		[ -f /mnt/mod/ctrl/configs/led_launch.cfg ] && cat /mnt/mod/ctrl/configs/led_launch.cfg > /sys/class/power_supply/axp2202-battery/work_led
	if $CMD > /dev/null 2>&1; then
		if [ -f "$ACT" ]; then
			if  ! sh $ACT > /dev/null 2>&1; then
	#           sleep 5
				echo "[allenapp] exe app fail ..."
			fi
	#        rm -f "$ACT"
		fi
	else
		sleep 30
	fi
}

function debug_app_scheduling()
{
	if $CMD > /dev/console 2>&1; then
		if [ -f "$ACT" ]; then
			if  ! sh $ACT > /dev/console 2>&1; then
	#           sleep 5
				echo "[allenapp] exe app fail ..." > /dev/console
			fi
	#        rm -f "$ACT"
		fi
	else
		sleep 30
	fi
}

if [ -f "/mnt/data/debug.ini" ];then
	debug_app_scheduling
elif [ -f "/mnt/sdcard/debug.ini" ];then
	debug_app_scheduling
else
	app_scheduling
fi
